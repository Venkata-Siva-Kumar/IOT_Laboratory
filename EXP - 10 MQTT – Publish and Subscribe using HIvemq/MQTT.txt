#include <Arduino.h>
#include <WiFi.h>
#include <PubSubClient.h>

#include "DHT.h" 
#define DHTPIN 4  
#define DHTTYPE DHT22   // DHT 11
DHT dht(DHTPIN, DHTTYPE);

WiFiClient wifiClient;
PubSubClient mqttClient(wifiClient); 

 

const char* ssid = "XXXXXX";
const char* password = "XXXXXXX";

char *mqttServer = "broker.hivemq.com";
int mqttPort = 1883;

void setupMQTT() 
{
	mqttClient.setServer(mqttServer, mqttPort);
	mqttClient.setCallback(callback);
}

void reconnect() 
{
	Serial.println("Connecting to MQTT Broker...");
	while (!mqttClient.connected()) 
	{
		Serial.println("Reconnecting to MQTT Broker..");
		String clientId = "ESP32Client-";
		clientId += String(random(0xffff), HEX);
      
		if (mqttClient.connect(clientId.c_str())) 
		{
			Serial.println("Connected.");
			mqttClient.subscribe("esp32/roomtemperature");
			delay(2000);
			mqttClient.subscribe("esp32/relativehumidity");
			delay(1000);
		}      
	}
}

void setup() 
{
	Serial.begin(115200);
	dht.begin();
	WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) 
	{
		delay(500);
		Serial.print(".");
    }
    Serial.println("");
    Serial.println("Connected to Wi-Fi");
	setupMQTT();
}


void loop() 
{
	if (!mqttClient.connected())
		reconnect();
		
	mqttClient.loop();
	long now = millis();
	long previous_time = 0;
  
	if (now - previous_time > 1000) 
	{
		previous_time = now;
		float humidity = dht.readHumidity();
		float temp = dht.readTemperature();  

		if (isnan(humidity) || isnan(temp)) 
		{
			Serial.println(F("Failed to read from DHT sensor!"));
			return;
		}
		char tempString[8];
		char humString[8];
		
		dtostrf(temp, 1, 2, tempString);
		
		//Serial.println(tempString);
		mqttClient.publish("esp32/roomtemperature",tempString);
		Serial.println("Temperature published");
		dtostrf(humidity, 1, 2, humString);
		
		//Serial.println(humString);
		mqttClient.publish("esp32/relativehumidity",humString);
		Serial.println("Humidity Published val");
	}
}

void callback(char* topic, byte* message, unsigned int length) 
{
	Serial.println("Callback - ");
	Serial.println("Message:");
	for (int i = 0; i < length; i++) 
	{
		Serial.print((char)message[i]);
		//delay(100);
	}
	delay(1000)
}
